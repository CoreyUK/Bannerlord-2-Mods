# Strategic Campaign AI - user settings
#
# Every line is  Key = Value.  '#' and '//' start a comment.
# Keys match the field names in StrategicAiTuning and are case-insensitive.
# Unknown keys and bad values are ignored, so this file can never break startup.
# Changes take effect when the game is restarted.
#
# Everything below is commented out and shows the built-in default.

# ---------------------------------------------------------------- master switches

# Direct move orders (army roles, staging, interception).
#
# OFF by default. The vanilla AI clears our move orders on its own schedule, so
# this layer ends up in a tug of war with it, and in testing the corrected
# scoring ALONE produced about three times as many successful sieges as it did
# with this enabled. Turn it on only if you want the mod steering armies
# directly and are willing to watch for armies that never reach their target.
# EnableStrategicOrders = false

# AI lords leaving extra troops to a garrison that is genuinely under threat.
# Now answered through SettlementGarrisonModel.FindNumberOfTroopsToLeaveToGarrison,
# so the engine performs the transfer. The old version mutated rosters directly
# and was the cause of a hard crash; nothing touches a roster now.
# EnableGarrisonReinforcement = true

# Armies abandoning a siege they are clearly losing.
# Now done by scoring a hopeless siege near zero so the vanilla AI lifts it
# itself, rather than forcing it through a private engine method.
# EnableSiegeRetreat = true

# Write decisions to the rgl log (Documents\Mount and Blade II Bannerlord\logs).
# VerboseLogging = false

# ------------------------------------------------------------------ score bounds
#
# The most important settings in the file. All the individual multipliers below
# are combined and then clamped into these bands. Without the clamp the penalties
# compound until no enemy settlement looks worth attacking, which is what makes
# armies sit in their own towns and patrol in circles.
#
# If armies still look too passive, raise MinOffenseMultiplier and lower
# MaxDefenseMultiplier. Setting both offence bounds to 1.0 disables offensive
# suppression outright and leaves vanilla siege scoring untouched.

# MinOffenseMultiplier = 0.45
# MaxOffenseMultiplier = 3.0
# MinDefenseMultiplier = 0.65
# MaxDefenseMultiplier = 1.85

# ------------------------------------------------------------------ AI cadence

# How often the strategic layer re-plans, in campaign hours.
# StrategicUpdateIntervalHours = 4

# Minimum hours between two strategic move orders for the same army.
# Lower values make the AI more reactive but risk it jittering.
# ArmyOrderMinIntervalHours = 12

# How long an army sticks with an objective before reconsidering it.
# ObjectiveCommitmentHours = 36

# Bonus applied to the objective an army already holds. This hysteresis is what
# stops two armies swapping targets forever. Raise it if armies still waver.
# CurrentObjectiveStickinessMultiplier = 1.4

# How long a failed target is avoided, in hours.

# ------------------------------------------------------------ persistent memory
#
# Carry siege failures, lord recovery cooldowns, army objectives, war goals and
# lost-fief claims across save/load. Only dictionaries of primitives are written,
# through a registered save definer.
#
# The save definer is always registered regardless of this setting, so turning
# persistence off can never make an existing save unloadable.

# EnablePersistentMemory = true

# Enemy strength near the capital that switches a kingdom to DefendCapitalRegion.
